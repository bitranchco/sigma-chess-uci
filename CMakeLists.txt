cmake_minimum_required(VERSION 3.10)
project(SigmaChessUCI C)

# Set C11 standard
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Compiler flags
add_compile_options(
    -Wall
    -Wextra
)

# Build-specific flags
# Debug: enables assertions, slower but catches bugs
# Release: disables assertions (NDEBUG), full optimization
if(CMAKE_BUILD_TYPE MATCHES Debug)
    add_compile_options(-g -O0)
    message(STATUS "Debug build: Assertions ENABLED, optimizations OFF")
else()
    add_compile_options(-O3 -g -DNDEBUG)
    # LTO when the compiler supports it (Tigerbrew gcc-7 ships without LTO, hence the check).
    include(CheckCCompilerFlag)
    check_c_compiler_flag("-flto" SIGMA_HAS_LTO)
    if(SIGMA_HAS_LTO)
        add_compile_options(-flto)
        add_link_options(-flto)
    endif()
    # Pin to apple-m1: subset of all later Apple Silicon, so runs on any M-series host.
    if(APPLE AND CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
        add_compile_options(-mcpu=apple-m1)
        add_link_options(-mcpu=apple-m1)
    endif()
    message(STATUS "Release build: Assertions DISABLED, optimizations ON, LTO=${SIGMA_HAS_LTO}")
endif()

# Syzygy prober (vendored Fathom) plus its bridge, in its own directory so it doesn't inherit
# Sigma's include directories or the General.h force-include below: Sigma's macros (white,
# black, file, rank, ...) would break Fathom, and src/Misc/time.h would shadow <time.h>.
add_subdirectory(src/Syzygy)

# Include directories
include_directories(
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/src/DataStructures
    ${CMAKE_SOURCE_DIR}/src/Misc
    ${CMAKE_SOURCE_DIR}/src/Evaluation
    ${CMAKE_SOURCE_DIR}/src/MoveGeneration
    ${CMAKE_SOURCE_DIR}/src/Searching
    ${CMAKE_SOURCE_DIR}/src/Syzygy
)

# Force-include General.h in all compilation units
add_compile_options(-include ${CMAKE_SOURCE_DIR}/src/General.h)

# Source files
set(SOURCES
    src/uci_main.c
    src/uci_bridge.c
    src/Engine.c
    src/DataStructures/Board.c
    src/DataStructures/Attack.c
    src/DataStructures/Move.c
    src/MoveGeneration/PerformMove.c
    src/MoveGeneration/MoveGen.c
    src/Searching/Search.c
    src/Searching/SearchMisc.c
    src/Searching/NodeSearch.c
    src/Searching/Threats.c
    src/Searching/Selection.c
    src/Searching/TransTables.c
    src/Evaluation/Evaluate.c
    src/Evaluation/PieceVal.c
    src/Misc/Time.c
    src/Misc/HashCode.c
    src/Misc/Tablebase.c
)

# Unity build option - compile all sources as single translation unit
option(UNITY_BUILD "Enable unity build (single object compilation)" ON)

if(UNITY_BUILD)
    # Create a single unity source file
    set(UNITY_SOURCE ${CMAKE_BINARY_DIR}/unity.c)

    # Generate unity.c that includes all source files
    file(WRITE ${UNITY_SOURCE} "// Unity build - single compilation unit\n")
    file(APPEND ${UNITY_SOURCE} "// Generated by CMake\n\n")
    foreach(source ${SOURCES})
        file(APPEND ${UNITY_SOURCE} "#include \"${CMAKE_SOURCE_DIR}/${source}\"\n")
    endforeach()

    # Create executable from unity source
    add_executable(sigma-uci ${UNITY_SOURCE})

    # Mark as unity build in properties
    set_target_properties(sigma-uci PROPERTIES
        UNITY_BUILD ON
        UNITY_BUILD_MODE BATCH
    )
else()
    # Standard build - separate compilation units
    add_executable(sigma-uci ${SOURCES})
endif()

# Link math library
target_link_libraries(sigma-uci fathom m)

# Install target
install(TARGETS sigma-uci DESTINATION bin)

# Print build configuration
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
message(STATUS "C compiler: ${CMAKE_C_COMPILER}")
message(STATUS "Unity build: ${UNITY_BUILD}")
if(UNITY_BUILD)
    message(STATUS "Unity source: ${UNITY_SOURCE}")
endif()

# ----------------------------------------------------------------------------
# Symmetry tests for PerformMove / RetractMove
# Reuses the unity-build pattern: compiles all engine sources together with
# the test sources into a single TU. Independent of src/tests/Makefile.
# ----------------------------------------------------------------------------
set(SYMMETRY_TEST_UNITY ${CMAKE_BINARY_DIR}/symmetry_unity.c)
file(WRITE ${SYMMETRY_TEST_UNITY} "// Symmetry test unity build - generated by CMake\n")
foreach(source ${SOURCES})
    if(NOT ${source} MATCHES "uci_main\\.c$")
        file(APPEND ${SYMMETRY_TEST_UNITY} "#include \"${CMAKE_SOURCE_DIR}/${source}\"\n")
    endif()
endforeach()
file(APPEND ${SYMMETRY_TEST_UNITY} "#include \"${CMAKE_SOURCE_DIR}/src/tests/symmetry/state_snapshot.c\"\n")
file(APPEND ${SYMMETRY_TEST_UNITY} "#include \"${CMAKE_SOURCE_DIR}/src/tests/symmetry/test_perform_move_symmetry.c\"\n")

add_executable(test_symmetry ${SYMMETRY_TEST_UNITY})
target_include_directories(test_symmetry PRIVATE ${CMAKE_SOURCE_DIR}/src/tests/symmetry)
target_link_libraries(test_symmetry fathom m)
